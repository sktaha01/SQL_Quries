9. Count the orders of each of the salespeople and output the results in descending order.

select s.snum  ,s.sname,  count(o.snum) No_Of_Orders
from salespeople s, orders o 
where s.snum =o.snum
group by s.snum
order by count(o.snum) desc;

select s.snum  ,s.sname,  count(o.snum) No_Of_Orders
